// PRIVATE MODEL. A sign-in identity, not a member. Never embedded.
// memberId is set by the system, in the same transaction that makes the account ACTIVE when
// the email is verified. An UNVERIFIED account has no member and no password.
model Account {
  id                   String        @id @default(cuid())
  name                 String
  email                String        @unique // stored lowercase and trimmed
  passwordHash         String?       // empty until the person uses their verification link and sets one
  sessionVersion       Int           @default(0) // add one to end every other session
  status               AccountStatus @default(UNVERIFIED)
  emailVerifiedAt      DateTime?
  phone                String        // required at sign-up; international digits only; copied to Member at verification. The person can correct it on the verification page
  claimsExistingMember Boolean       // the answer to "Already a member at the gym?"
  memberId             String?       @unique
  createdAt            DateTime      @default(now())
  updatedAt            DateTime      @updatedAt

  member               Member?       @relation(fields: [memberId], references: [id])
  tokens               EmailToken[]

  @@index([status, createdAt])
}

// PRIVATE MODEL. Single-use links for the two account emails. Stores a hash, never the token.
// Belongs to exactly one of an Account or a Staff row. A CHECK constraint in the raw
// migration enforces that. Staff tokens are RESET only.
model EmailToken {
  id          String          @id @default(cuid())
  accountId   String?
  staffId     String?
  type        EmailTokenType
  tokenHash   String          @unique
  expiresAt   DateTime
  usedAt      DateTime?
  createdAt   DateTime        @default(now())

  account     Account?        @relation(fields: [accountId], references: [id], onDelete: Cascade)
  staff       Staff?          @relation(fields: [staffId], references: [id], onDelete: Cascade)

  @@index([accountId])
  @@index([staffId])
}

// PRIVATE MODEL. Operational counter for the wrong-password and wrong-check-in-code pauses.
// Not a member record. Rows older than 24 hours may be deleted (privacy.md).
// key is the lowercase email for the two login kinds.
model FailedAttempt {
  id         String      @id @default(cuid())
  kind       AttemptKind
  key        String
  createdAt  DateTime    @default(now())

  @@index([kind, key, createdAt])
}

// PRIVATE MODEL. Fetched by member ID only. Never embedded.
model Member {
  id                 String        @id @default(cuid())
  membershipId       String        @unique // SPT-XXXX, generated by the system, permanent
  name               String
  phone              String?       // international digits only; gym contact, not a sign-in field
  // Paid through the whole of this date (Africa/Lagos). Null means never paid. The member's plan
  // is worked out from this in src/server/plan/ and nowhere else. Nothing is written when it
  // runs out.
  paidUntil          DateTime?     @db.Date
  // The sign-up answer, copied from the account. The owner can change it (audited).
  claimsExistingMember Boolean     @default(false)
  openingBalanceSet  Boolean       @default(false)
  accessCardIssuedAt   DateTime?
  accessCardIssuedById String?
  createdAt          DateTime      @default(now())
  updatedAt          DateTime      @updatedAt

  accessCardIssuedBy Staff?        @relation("CardIssuer", fields: [accessCardIssuedById], references: [id])

  account            Account?

  attendance         Attendance[]
  ledgerEntries      LedgerEntry[]
  paymentAttempts    PaymentAttempt[]
  questionLogs       QuestionLog[]
  wrongAnswerReports WrongAnswerReport[]
  changes            MemberChange[]

  @@index([paidUntil])
}

// PRIVATE MODEL. Audit trail for paid-until, the existing-member answer and the access card.
// A change made by a person has an author. A change made by a verified payment has no author,
// source PAYMENT, and the payment attempt that caused it.
model MemberChange {
  id               String         @id @default(cuid())
  memberId         String
  field            MemberField
  oldValue         String
  newValue         String
  authorId         String?
  source           ChangeSource   @default(STAFF)
  paymentAttemptId String?        @unique
  createdAt        DateTime       @default(now())

  member           Member         @relation(fields: [memberId], references: [id], onDelete: Cascade)
  author           Staff?         @relation(fields: [authorId], references: [id], onDelete: Restrict)
  paymentAttempt   PaymentAttempt? @relation(fields: [paymentAttemptId], references: [id])

  @@index([memberId])
}

// Settings the owner edits in the app. monthly_price_naira is the price of one month, in whole
// naira. It is never taken from card text. No row exists until the owner sets one, and Pay
// stays closed until then (D27).

  key         String   @id
  intValue    Int
  updatedAt   DateTime @updatedAt
  updatedById String?

  updatedBy   Staff?   @relation("SettingEditor", fields: [updatedById], references: [id])
}

// Exactly one row may have active = true. Enforced by a partial unique index.
model Duty {
  id         String   @id @default(cuid())
  staffId    String
  since      DateTime @default(now())
  active     Boolean  @default(true)

  staff      Staff    @relation(fields: [staffId], references: [id])

  @@index([active])
}

Raw migration needed after the schema migration

CREATE UNIQUE INDEX one_active_duty ON "Duty" (active) WHERE active = true;


ALTER TABLE "EmailToken" ADD CONSTRAINT email_token_one_owner
  CHECK (("accountId" IS NULL) <> ("staffId" IS NULL));


Prisma cannot express a partial unique index or a CHECK constraint, so these run as a raw SQL migration. The CHECK makes every EmailToken belong to exactly one of an Account or a Staff row.
